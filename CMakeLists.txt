cmake_minimum_required(VERSION 3.22)
project(LinuxGPUControl VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network OpenGL DBus)
find_package(SQLite3 REQUIRED)
if(TARGET SQLite3::SQLite3)
 set(SQLITE_TARGET SQLite3::SQLite3)
else()
 set(SQLITE_TARGET SQLite::SQLite3)
endif()
find_package(Vulkan REQUIRED)
find_package(X11 REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(ARCHIVE REQUIRED IMPORTED_TARGET libarchive)
pkg_check_modules(DRM REQUIRED IMPORTED_TARGET libdrm)
pkg_check_modules(GTK3 REQUIRED IMPORTED_TARGET gtk+-3.0 pangocairo)
find_program(GLSLANG glslangValidator REQUIRED)
foreach(stage vert frag)
  add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/city.${stage}.spv
    COMMAND ${GLSLANG} -V ${CMAKE_CURRENT_SOURCE_DIR}/shaders/city.${stage} -o ${CMAKE_CURRENT_BINARY_DIR}/city.${stage}.spv
    DEPENDS shaders/city.${stage} VERBATIM)
endforeach()
set(COMPUTE_SPV)
foreach(kernel copy write read fp32 fp64 int32 int64)
 add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${kernel}.comp.spv COMMAND ${GLSLANG} -V ${CMAKE_CURRENT_SOURCE_DIR}/shaders/${kernel}.comp -o ${CMAKE_CURRENT_BINARY_DIR}/${kernel}.comp.spv DEPENDS shaders/${kernel}.comp VERBATIM)
 list(APPEND COMPUTE_SPV ${CMAKE_CURRENT_BINARY_DIR}/${kernel}.comp.spv)
endforeach()
add_custom_target(shaders DEPENDS ${CMAKE_CURRENT_BINARY_DIR}/city.vert.spv ${CMAKE_CURRENT_BINARY_DIR}/city.frag.spv ${COMPUTE_SPV})
add_library(gpu-core STATIC src/gpu.cpp src/wsl_gpu.cpp src/gpu_tuning.cpp src/gpu.h src/model.cpp src/model.h src/cpu_telemetry.cpp src/cpu_telemetry.h src/frame_log.cpp src/frame_log.h src/repository.cpp src/repository.h src/driver_packages.cpp src/driver_packages.h src/driver_stack.cpp src/driver_stack.h src/intel_specs.cpp src/intel_specs.h src/system_settings.cpp src/system_settings.h)
target_link_libraries(gpu-core PUBLIC Qt6::Core Qt6::DBus PkgConfig::DRM ${SQLITE_TARGET} ${CMAKE_DL_LIBS})
add_executable(linux-gpu-control src/main.cpp src/branding.cpp src/branding.h src/single_instance.cpp src/single_instance.h src/window.cpp src/window.h src/service.cpp src/service.h src/overlay.cpp src/benchmark.cpp src/benchmark.h src/codec.cpp src/codec.h src/codec_suite.cpp src/codec_suite.h src/window_tests.cpp src/window_tuning.cpp src/window_specs.cpp src/window_feedback.cpp src/neon_card.cpp src/neon_card.h src/window_updates.cpp src/window_drivers.cpp src/window_system.cpp src/update_manager.cpp src/update_manager.h)
qt_add_resources(linux-gpu-control bench-shaders PREFIX /shaders BASE ${CMAKE_CURRENT_BINARY_DIR}
  FILES ${CMAKE_CURRENT_BINARY_DIR}/city.vert.spv ${CMAKE_CURRENT_BINARY_DIR}/city.frag.spv ${COMPUTE_SPV})
qt_add_resources(linux-gpu-control branding PREFIX / FILES assets/gpu-pro.png docs/AVISO-DE-RESPONSABILIDADE.md LICENSE NOTICE docs/licenses/LGPL-3.0.txt docs/licenses/GPL-3.0.txt docs/licenses/LGPL-2.1.txt)
add_dependencies(linux-gpu-control shaders)
target_link_libraries(linux-gpu-control PRIVATE gpu-core Qt6::Widgets Qt6::Network Vulkan::Vulkan PkgConfig::ARCHIVE PkgConfig::GTK3 X11::X11)
add_executable(gpu-bench-worker src/render_worker.cpp)
add_dependencies(gpu-bench-worker shaders)
target_link_libraries(gpu-bench-worker PRIVATE gpu-core Qt6::Gui Qt6::OpenGL Vulkan::Vulkan)
qt_add_resources(gpu-bench-worker worker-shaders PREFIX /shaders BASE ${CMAKE_CURRENT_BINARY_DIR} FILES ${CMAKE_CURRENT_BINARY_DIR}/city.vert.spv ${CMAKE_CURRENT_BINARY_DIR}/city.frag.spv ${COMPUTE_SPV})
qt_add_resources(gpu-bench-worker worker-source PREFIX /source BASE ${CMAKE_CURRENT_SOURCE_DIR}/shaders FILES shaders/city.frag)
add_executable(gpu-clock-helper src/power_helper.cpp)
target_link_libraries(gpu-clock-helper PRIVATE gpu-core)
add_executable(gpu-video-worker src/video_worker.cpp src/codec.cpp src/codec_suite.cpp)
target_link_libraries(gpu-video-worker PRIVATE gpu-core)
add_executable(gpu-diagnostics src/diagnostics.cpp)
target_link_libraries(gpu-diagnostics PRIVATE gpu-core)
add_executable(core-checks src/checks.cpp)
target_link_libraries(core-checks PRIVATE gpu-core)
enable_testing()
add_test(NAME offline-core COMMAND core-checks)
foreach(target gpu-core linux-gpu-control gpu-clock-helper gpu-bench-worker gpu-diagnostics gpu-video-worker core-checks)
 target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
endforeach()
add_executable(gpu-system-worker src/system_worker.cpp)
target_link_libraries(gpu-system-worker PRIVATE gpu-core Qt6::Core Qt6::Network)
install(TARGETS linux-gpu-control gpu-bench-worker gpu-diagnostics gpu-video-worker gpu-system-worker RUNTIME DESTINATION bin)
install(TARGETS gpu-clock-helper RUNTIME DESTINATION libexec/linux-gpu-control)

add_test(NAME reject-power-write COMMAND gpu-clock-helper 0000:01:00.0 power 225000)
set_tests_properties(reject-power-write PROPERTIES WILL_FAIL TRUE)
add_test(NAME reject-voltage-write COMMAND gpu-clock-helper 0000:01:00.0 voltage 1000)
set_tests_properties(reject-voltage-write PROPERTIES WILL_FAIL TRUE)

add_executable(update-checks src/update_checks.cpp src/update_manager.cpp src/update_manager.h)
target_link_libraries(update-checks PRIVATE gpu-core Qt6::Core Qt6::Network PkgConfig::ARCHIVE)
target_compile_options(update-checks PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME updater-backup-restore COMMAND update-checks)
add_test(NAME interface-layout COMMAND linux-gpu-control --ui-check)
set_tests_properties(interface-layout PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME tray-hidden-exit COMMAND linux-gpu-control --tray-exit-check)
set_tests_properties(tray-hidden-exit PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 5)

add_test(NAME hud-style-risk-interactions COMMAND linux-gpu-control --interaction-check)
set_tests_properties(hud-style-risk-interactions PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_STYLE_OVERRIDE=kvantum" TIMEOUT 10)

add_executable(instance-checks src/instance_checks.cpp src/single_instance.cpp src/single_instance.h)
target_link_libraries(instance-checks PRIVATE Qt6::Core Qt6::Network)
add_test(NAME single-instance-activation-recovery COMMAND instance-checks)
set_tests_properties(single-instance-activation-recovery PROPERTIES TIMEOUT 15)
add_test(NAME branding-transparency-colors COMMAND linux-gpu-control --branding-check)
set_tests_properties(branding-transparency-colors PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen" TIMEOUT 10)

add_executable(tuning-checks src/tuning_checks.cpp)
target_link_libraries(tuning-checks PRIVATE gpu-core)
target_compile_options(tuning-checks PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME manual-tuning-and-frame-metrics COMMAND tuning-checks)

add_executable(telemetry-checks src/telemetry_checks.cpp src/service.cpp src/service.h)
target_link_libraries(telemetry-checks PRIVATE gpu-core Qt6::Network)
target_compile_options(telemetry-checks PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME frame-source-live-and-stale COMMAND telemetry-checks)
set_tests_properties(frame-source-live-and-stale PROPERTIES TIMEOUT 10)

add_executable(wsl-checks src/wsl_checks.cpp)
target_link_libraries(wsl-checks PRIVATE gpu-core)
target_compile_options(wsl-checks PRIVATE -Wall -Wextra -Wpedantic)
add_test(NAME wsl-nvml-readonly-discovery COMMAND wsl-checks)
set_tests_properties(wsl-nvml-readonly-discovery PROPERTIES SKIP_RETURN_CODE 77 TIMEOUT 10)
